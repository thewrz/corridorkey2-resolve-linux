cmake_minimum_required(VERSION 3.20)
project(CorridorKeyResolve VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(OFX_SDK "/opt/resolve/Developer/OpenFX" CACHE PATH "Resolve OpenFX SDK")
enable_testing()
add_library(ck_plugin_runtime INTERFACE)
# Resolve loads libraries exporting older, ABI-incompatible C++ filesystem symbols.
# Keep our runtime private; the host/plugin contract contains only OFX C types.
target_link_options(ck_plugin_runtime INTERFACE
    -static-libstdc++ -static-libgcc
    -Wl,--exclude-libs,libstdc++.a:libgcc.a:libgcc_eh.a -Wl,-Bsymbolic)
add_library(abi_probe MODULE tests/abi_probe.cpp)
target_link_libraries(abi_probe PRIVATE ck_plugin_runtime)
add_executable(abi_host tests/abi_host.cpp)
target_link_libraries(abi_host PRIVATE dl)
target_link_options(abi_host PRIVATE -Wl,--export-dynamic)
add_test(NAME host_runtime_isolation COMMAND abi_host $<TARGET_FILE:abi_probe>)
add_library(ck_image STATIC native/image.cpp)
target_include_directories(ck_image PUBLIC native)
target_compile_options(ck_image PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(native_tests tests/native_tests.cpp)
target_link_libraries(native_tests PRIVATE ck_image)
add_test(NAME native_image_contracts COMMAND native_tests)
add_library(ck_bridge STATIC native/bridge.cpp)
target_include_directories(ck_bridge PUBLIC native vendor)
target_compile_options(ck_bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(bridge_tests tests/bridge_tests.cpp)
target_link_libraries(bridge_tests PRIVATE ck_bridge)
add_test(NAME worker_lifecycle COMMAND bridge_tests)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(ck_shot STATIC native/shot.cpp)
target_link_libraries(ck_shot PUBLIC ck_bridge ck_image OpenSSL::Crypto)
target_compile_options(ck_shot PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(shot_tests tests/shot_tests.cpp)
target_link_libraries(shot_tests PRIVATE ck_shot)
add_test(NAME shot_cache_contracts COMMAND shot_tests)

if(NOT EXISTS "${OFX_SDK}/Support/include/ofxsImageEffect.h")
  message(FATAL_ERROR "Install Resolve's Developer/OpenFX SDK or set OFX_SDK")
endif()
set(SUPPORT_SOURCES ofxsCore ofxsImageEffect ofxsInteract ofxsLog ofxsMultiThread
    ofxsParams ofxsProperty ofxsPropertyValidation)
list(TRANSFORM SUPPORT_SOURCES PREPEND "${OFX_SDK}/Support/Library/")
list(TRANSFORM SUPPORT_SOURCES APPEND ".cpp")
add_library(ofx_support STATIC ${SUPPORT_SOURCES})
target_include_directories(ofx_support PUBLIC "${OFX_SDK}/Support/include" "${OFX_SDK}/OpenFX-1.4/include")
target_compile_definitions(ofx_support PRIVATE NDEBUG)
target_compile_options(ofx_support PRIVATE -fvisibility=hidden)
add_executable(host_tests tests/host_tests.cpp)
target_include_directories(host_tests PRIVATE native)
target_link_libraries(host_tests PRIVATE ofx_support)
add_test(NAME optional_host_notifications COMMAND host_tests)
file(READ native/parameters.json PARAMETER_SCHEMA)
configure_file(native/schema.hpp.in generated/schema.hpp @ONLY)
add_library(CorridorKey2 MODULE native/plugin.cpp native/parameters.cpp)
target_include_directories(CorridorKey2 PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(CorridorKey2 PRIVATE ofx_support ck_shot ck_plugin_runtime dl pthread)
target_compile_options(CorridorKey2 PRIVATE -Wall -Wextra -Wpedantic -Werror -fvisibility=hidden)
target_link_options(CorridorKey2 PRIVATE "-Wl,--version-script=${CMAKE_SOURCE_DIR}/native/exports.map")
set_property(TARGET CorridorKey2 APPEND PROPERTY LINK_DEPENDS "${CMAKE_SOURCE_DIR}/native/exports.map")
set(BUNDLE "${CMAKE_BINARY_DIR}/CorridorKey2.ofx.bundle")
set_target_properties(CorridorKey2 PROPERTIES PREFIX "" SUFFIX ".ofx"
    LIBRARY_OUTPUT_DIRECTORY "${BUNDLE}/Contents/Linux-x86-64")
add_executable(ofx_contract_tests tests/ofx_contract_tests.cpp)
target_include_directories(ofx_contract_tests PRIVATE "${OFX_SDK}/OpenFX-1.4/include")
target_link_libraries(ofx_contract_tests PRIVATE dl)
target_compile_options(ofx_contract_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME ofx_context_contracts COMMAND ofx_contract_tests $<TARGET_FILE:CorridorKey2>)
file(MAKE_DIRECTORY "${BUNDLE}/Contents/Resources")
configure_file(native/runtime.json.in "${BUNDLE}/Contents/Resources/runtime.json" @ONLY)
add_custom_target(bundle_resources ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_SOURCE_DIR}/worker" "${BUNDLE}/Contents/Resources/worker"
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_SOURCE_DIR}/licenses" "${BUNDLE}/Contents/Resources/licenses"
    COMMAND ${CMAKE_COMMAND} -E copy
        "${CMAKE_SOURCE_DIR}/THIRD_PARTY_NOTICES.md" "${BUNDLE}/Contents/Resources/"
    DEPENDS CorridorKey2)
